║                    Walmart PDF 数据库设计 V2.0 总览                          ║
║                        动态板块 + 自动低频字段合并                            ║
╚══════════════════════════════════════════════════════════════════════════════╝

┌─ 设计目标 ──────────────────────────────────────────────────────────────────┐
│                                                                               │
│  1. 一个 PDF = 一条 statement 记录                                             │
│  2. 板块动态化（支持未来增加新板块，无需修改 schema）                         │
│  3. 低频字段（频率 < 2）自动合并到 {section_name}_其他                       │
│  4. 支持 SQL 查询（可聚合、可统计、可过滤）                                    │
│  5. 最小化表数（仅 2 个主表），最简化查询复杂性                               │
│                                                                               │
└───────────────────────────────────────────────────────────────────────────────┘

┌─ 数据流向 ──────────────────────────────────────────────────────────────────┐
│                                                                               │
│   PDF 文件                                                                    │
│     │                                                                         │
│     ├─→ parse_pdf_direct()                                                   │
│     │     │                                                                  │
│     │     ├─→ Step 1: 灰度 + 水平分割                                         │
│     │     ├─→ Step 2: 分割左右部分                                            │
│     │     ├─→ Step 3: OCR 识别                                               │
│     │     ├─→ Step 4: 文本行合并                                              │
│     │     └─→ Step 5: jg_structured_data() ← 结构化提取                      │
│     │                                                                        │
│     ├─→ jg_structured_data() 返回值                                          │
│     │   {                                                                   │
│     │     "sections": {                                                     │
│     │       "header": [items],                                             │
│     │       "销售": [items],                                                │
│     │       "退款": [items],                                                │
│     │       ...                                                            │
│     │     },                                                               │
│     │     "metadata": {...}                                               │
│     │   }                                                                   │
│     │                                                                        │
│     └─→ StructuredDataImporter.import_jg_data()                             │
│           │                                                                 │
│           ├─ 步骤 1: 创建 statement 记录 (来自 header)                        │
│           │   INSERT INTO statements (pdf_name, period, payment, ...)       │
│           │   → statement_id                                               │
│           │                                                                │
│           ├─ 步骤 2: 遍历每个 section                                        │
│           │   FOR section_name, items IN sections:                         │
│           │       → 将 items 转换为 dict                                    │
│           │                                                                │
│           ├─ 步骤 3: 识别并合并低频字段                                      │
│           │   高频字段 (频率≥2) → 顶级保存                                   │
│           │   低频字段 (频率<2) → {section_name}_其他 (JSON)                │
│           │                                                                │
│           └─ 步骤 4: 写入 section_data 记录                                  │
│               INSERT INTO section_data (statement_id, section_name, data)   │
│                                                                             │
│     ↓ 结果                                                                   │
│                                                                             │
│   数据库：                                                                    │
│   ├─ statements: 1 条记录 (PDF 的基本信息)                                   │
│   └─ section_data: N 条记录 (N = 板块数)                                    │
│                                                                               │
└───────────────────────────────────────────────────────────────────────────────┘

┌─ 表结构 ────────────────────────────────────────────────────────────────────┐
│                                                                               │
│  ┌─────────────────────────────────────────────────────────────────┐         │
│  │ statements (语句主表)                                             │         │
│  ├──────────────────────────────┬──────────────────────────────────┤         │
│  │ 字段                           │ 说明                             │         │
│  ├──────────────────────────────┼──────────────────────────────────┤         │
│  │ id (PK)                      │ 主键                              │         │
│  │ pdf_name (UNIQUE)            │ PDF 文件名                        │         │
│  │ statement_period             │ 统计区间 (来自 header 板块)      │         │
│  │ payment_to_you               │ 应付金额 (来自 header 板块)      │         │
│  │ opening_balance              │ 期初余额 (来自 header 板块)      │         │
│  │ reserve_fund                 │ 备用金 (来自 header 板块)        │         │
│  │ pending_payment              │ 回款等待 (来自 header 板块)      │         │
│  │ created_at, updated_at       │ 审计时间戳                        │         │
│  └──────────────────────────────┴──────────────────────────────────┘         │
│                                  ↑                                           │
│                                  │ 1:N 关系                                 │
│                                  │                                           │
│  ┌──────────────────────────────────────────────────────────────────┐        │
│  │ section_data (板块数据从表)                                        │        │
│  ├─────────────────────────────────┬───────────────────────────────┤        │
│  │ 字段                              │ 说明                           │        │
│  ├─────────────────────────────────┼───────────────────────────────┤        │
│  │ id (PK)                         │ 主键                            │        │
│  │ statement_id (FK)               │ 关联 statements                 │        │
│  │ section_name                    │ 板块名 (销售/退款/调整/etc)    │        │
│  │ data (JSON)                     │ 该板块的所有字段                │        │
│  │ created_at                      │ 创建时间                        │        │
│  │ UNIQUE(statement_id, section)   │ 约束：一个 PDF 每个板块一条    │        │
│  └─────────────────────────────────┴───────────────────────────────┘        │
│                                                                               │
│  辅助表：                                                                      │
│  ├─ field_frequency: 字段频率映射 (用于导入时识别低频字段)                   │
│  └─ db_config: 配置表                                                        │
│                                                                               │
└───────────────────────────────────────────────────────────────────────────────┘

┌─ JSON 数据结构 ─────────────────────────────────────────────────────────────┐
│                                                                               │
│  section_data.data 字段的内容示例 (销售板块)：                                │
│                                                                               │
│  {                                                                           │
│    "产品价格": 2000.00,              ← 频率 100%, 顶级                       │
│    "运输": 50.00,                    ← 频率 100%, 顶级                       │
│    "已收税净额": 25.00,               ← 频率 100%, 顶级                       │
│    "净佣金": 100.00,                 ← 频率 100%, 顶级                       │
│    "扣缴税款净额": 25.00,             ← 频率 100%, 顶级                       │
│    "总计": 2050.00,                  ← 频率 100%, 顶级                       │
│    "WFS运输退款": 20.00,              ← 频率 83%, 顶级                        │
│    "WFS运输税退款": 5.00,             ← 频率 83%, 顶级                        │
│    "T沃尔玛出资的节余": 10.00,        ← 频率 83%, 顶级                        │
│    "销售_其他": {                     ← 频率 <2 的字段合并到此               │
│      "其他税款(费用)": 8.00           ← 频率 17%, 合并                       │
│    }                                                                         │
│  }                                                                           │
│                                                                               │
└───────────────────────────────────────────────────────────────────────────────┘

┌─ 字段频率分布 ──────────────────────────────────────────────────────────────┐
│                                                                               │
│  统计数据 (基于 6 个 PDF)：                                                   │
│  ├─ 总字段数: 31                                                             │
│  ├─ 频率 ≥ 2 (直接存储): 19 个字段  ← 顶级存储在 JSON                        │
│  ├─ 频率 < 2 (合并): 12 个字段     ← {section_name}_其他 JSON                │
│  ├─ 频率分布:                                                               │
│  │   • 频率 6/6 (100%): 11 个字段                                            │
│  │   • 频率 5/6 (83%):  8 个字段                                             │
│  │   • 频率 4/6 (67%):  2 个字段                                             │
│  │   • 频率 3/6 (50%):  1 个字段                                             │
│  │   • 频率 2/6 (33%):  0 个字段                                             │
│  │   • 频率 1/6 (17%):  12 个字段 (合并)                                     │
│  └─ 8 个板块类型: header, 销售, 退款, 调整, 其他活动, WFS商品, WFS配送, footer│
│                                                                               │
└───────────────────────────────────────────────────────────────────────────────┘

┌─ 常见查询模式 ──────────────────────────────────────────────────────────────┐
│                                                                               │
│  查询 1: 获取一个 PDF 的完整数据                                              │
│  SELECT s.*, sd.section_name, sd.data                                        │
│  FROM statements s                                                           │
│  LEFT JOIN section_data sd ON s.id = sd.statement_id                         │
│  WHERE s.pdf_name = 'MP_01142025.pdf';                                       │
│                                                                               │
│  查询 2: 汇总销售额（按月）                                                   │
│  SELECT                                                                      │
│      s.statement_period,                                                    │
│      json_extract(sd.data, '$.产品价格') as sales_amount                    │
│  FROM statements s                                                           │
│  LEFT JOIN section_data sd ON s.id = sd.statement_id                         │
│  WHERE sd.section_name = '销售'                                              │
│  GROUP BY s.statement_period;                                               │
│                                                                               │
│  查询 3: 计算退款率                                                           │
│  SELECT                                                                      │
│      s.statement_period,                                                    │
│      CAST(json_extract(r.data, '$.总计') AS REAL) / 
│         CAST(json_extract(s.data, '$.总计') AS REAL) * 100 as refund_rate   │
│  FROM statements st                                                          │
│  LEFT JOIN section_data s ON st.id = s.statement_id AND s.section_name='销售'│
│  LEFT JOIN section_data r ON st.id = r.statement_id AND r.section_name='退款'│
│                                                                               │
│  查询 4: 查看低频字段                                                         │
│  SELECT s.pdf_name, sd.section_name, json_extract(sd.data, '$.xxx_其他')   │
│  FROM statements s                                                           │
│  LEFT JOIN section_data sd ON s.id = sd.statement_id                         │
│  WHERE json_extract(sd.data, '$.xxx_其他') IS NOT NULL;                     │
│                                                                               │
└───────────────────────────────────────────────────────────────────────────────┘

┌─ 设计优势对比 ──────────────────────────────────────────────────────────────┐
│                                                                               │
│                      V1 (已废弃)          V2 (当前)                          │
│  ├─ 表数量              7 个表              2 个表         ← 简化 70%         │
│  ├─ 主键关系            复杂多层 FK         简单 1:N       ← 易于理解         │
│  ├─ 新增板块            需创建新表          无需修改       ← 完全动态化       │
│  ├─ 新增字段            需修改表结构        自动 JSON      ← 零修改成本       │
│  ├─ 查询复杂性          多重 JOIN           单表+JSON      ← 查询简化 60%    │
│  ├─ NULL 列数           大量（每表多列）    最少           ← 存储效率高       │
│  ├─ 维护成本            高（多表同步）      低             ← 降低 80%        │
│  └─ 应对变化            困难                灵活           ← 未来适应强       │
│                                                                               │
└───────────────────────────────────────────────────────────────────────────────┘
